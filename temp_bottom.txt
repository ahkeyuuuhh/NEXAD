  return (
    <View style={styles.mainContainer}>
      {/* Top App Bar */}
      <View style={styles.topNav}>
        <TouchableOpacity onPress={() => navigation.goBack()} style={styles.navButton}>
          <Ionicons name="arrow-back" size={24} color={C.ink1} />
        </TouchableOpacity>
        <TouchableOpacity style={styles.navButton}>
          <Ionicons name="ellipsis-vertical" size={24} color={C.ink1} />
        </TouchableOpacity>
      </View>

      <ScrollView style={styles.scrollContent} contentContainerStyle={styles.scrollInner}>
        {/* Top Info */}
        <Text style={styles.dueDateText}>
          {bin.deadline ? `${isDeadlinePassed ? 'Work cannot be handed in after the due date' : `Due ${new Date(bin.deadline).toLocaleString('en-US', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit', hour12: false })}`}` : 'No due date'}
        </Text>
        {isDeadlinePassed && bin.deadline && (
           <Text style={styles.deadlineWarning}>Work cannot be handed in after the due date</Text>
        )}

        {/* Title */}
        <Text style={styles.assignmentTitle}>{bin.title}</Text>
        <Text style={styles.pointsText}>100 points</Text>

        <View style={styles.divider} />

        {/* Add class comment */}
        <TouchableOpacity style={styles.commentRow} onPress={openComments}>
          <Ionicons name="chatbubble-ellipses-outline" size={20} color="#0b57d0" />
          <Text style={styles.commentText}>Add class comment</Text>
        </TouchableOpacity>

        <View style={styles.divider} />

        {/* Description / Instructions */}
        {bin.description ? (
          <Text style={styles.descriptionText}>{bin.description}</Text>
        ) : null}

        {/* Teacher Feedback Alert (placed inline before the bottom sheet) */}
        {submission && reviewStatus && (
          <View style={[styles.teacherFeedbackBox, reviewStatus === 'approved' ? styles.feedbackApproved : reviewStatus === 'revised' ? styles.feedbackRevised : {}]}>
            <View style={{flexDirection: 'row', alignItems: 'center', gap: 6, marginBottom: 4}}>
              <Ionicons name={statusCfg?.icon as any} size={18} color={statusCfg?.color || C.ink1} />
              <Text style={[styles.feedbackTitle, {color: statusCfg?.color || C.ink1}]}>{statusCfg?.label}</Text>
            </View>
            
            {reviewStatus === 'for_consultation' || reviewStatus === 'consultation_requested' ? (
              <View>
                {consultationRequest && ['pending', 'ai_processing', 'awaiting_teacher'].includes(consultationRequest.status) ? (
                   <Text style={styles.feedbackHint}>Consultation booked. Awaiting teacher's response.</Text>
                ) : consultationRequest?.status === 'accepted' ? (
                   <Text style={styles.feedbackHint}>
                     Consultation Approved for {consultationRequest.scheduled_start_time ? new Date(consultationRequest.scheduled_start_time).toLocaleString() : 'TBD'}.
                   </Text>
                ) : (
                   <TouchableOpacity style={styles.requestConsultBtn} onPress={requestConsultation}>
                      <Text style={styles.requestConsultText}>Request Consultation</Text>
                   </TouchableOpacity>
                )}
              </View>
            ) : reviewStatus === 'revised' ? (
               <Text style={styles.feedbackHint}>Please make the necessary revisions and resubmit.</Text>
            ) : null}
          </View>
        )}
      </ScrollView>

      {/* Attach Menu Popover Background Scrim */}
      {showAttachMenu && (
        <TouchableOpacity style={styles.scrim} activeOpacity={1} onPress={() => setShowAttachMenu(false)} />
      )}

      {/* Your Work Bottom Sheet Area */}
      <View style={styles.bottomWorkContainer}>
        
        {/* Floating Attach Menu (visible when adding work) */}
        {showAttachMenu && (
           <View style={styles.attachMenuPopover}>
             <TouchableOpacity style={styles.attachMenuItem} onPress={pickDocumentAttachment}>
                <Ionicons name="document-text-outline" size={20} color={C.ink2} />
                <Text style={styles.attachMenuItemText}>File (PDF or DOCX)</Text>
             </TouchableOpacity>
             <TouchableOpacity style={styles.attachMenuItem} onPress={pickImageAttachment}>
                <Ionicons name="image-outline" size={20} color={C.ink2} />
                <Text style={styles.attachMenuItemText}>Images</Text>
             </TouchableOpacity>
           </View>
        )}

        {showAnalyzePrompt && selectedFile && !isImageFile(selectedFile.name, selectedFile.mimeType) && (
          <View style={styles.analyzePromptCard}>
            <Text style={styles.analyzePromptTitle}>Academic Integrity Check</Text>
            <Text style={styles.analyzePromptText} numberOfLines={1}>{selectedFile.name}</Text>
            <View style={styles.analyzePromptActions}>
              <TouchableOpacity style={styles.analyzePromptLaterBtn} onPress={() => setShowAnalyzePrompt(false)}>
                <Text style={styles.analyzePromptLaterText}>Later</Text>
              </TouchableOpacity>
              <TouchableOpacity
                style={styles.analyzePromptNowBtn}
                onPress={() => {
                  setShowAnalyzePrompt(false);
                  performPlagiarismCheck(selectedFile);
                }}
              >
                <Text style={styles.analyzePromptNowText}>Analyze Now</Text>
              </TouchableOpacity>
            </View>
          </View>
        )}

        <View style={styles.dragIndicator} />
        
        <View style={styles.workHeader}>
          <Text style={styles.workTitle}>Your work</Text>
          <Text style={[styles.workStatus, submission ? styles.statusTurnedIn : (isDeadlinePassed ? styles.statusMissing : styles.statusAssigned)]}>
            {submission ? 'Turned in' : (isDeadlinePassed ? 'Missing' : 'Assigned')}
          </Text>
        </View>

        {/* Selected or Submitted File Card */}
        {submission ? (
           <TouchableOpacity style={styles.fileCard} onPress={handleViewSubmission}>
             <View style={styles.fileIconBox}><Ionicons name="document-text" size={20} color="#D93025" /></View>
             <View style={styles.fileDetailsBox}>
               <Text style={styles.fileNameText} numberOfLines={1}>{submission.file_name}</Text>
             </View>
           </TouchableOpacity>
        ) : selectedFile ? (
           <View style={styles.fileCard}>
             <View style={styles.fileIconBox}><Ionicons name="document-text" size={20} color="#0B57D0" /></View>
             <View style={styles.fileDetailsBox}>
               <Text style={styles.fileNameText} numberOfLines={1}>{selectedFile.name}</Text>
             </View>
             <TouchableOpacity style={{padding: 12}} onPress={() => setSelectedFile(null)}>
               <Ionicons name="close" size={20} color="#5F6368" />
             </TouchableOpacity>
           </View>
        ) : null}

        {/* Buttons */}
        {canSubmit && !selectedFile && !submission ? (
           <TouchableOpacity style={styles.addWorkBtn} onPress={() => setShowAttachMenu(true)}>
              <Ionicons name="add" size={20} color="#FFF" />
              <Text style={styles.addWorkText}>Add work</Text>
           </TouchableOpacity>
        ) : null}

        {canSubmit && selectedFile ? (
           <TouchableOpacity style={styles.turnInBtn} onPress={handleSubmit} disabled={uploading}>
              {uploading ? <ActivityIndicator color="#fff" /> : <Text style={styles.turnInText}>Turn in</Text>}
           </TouchableOpacity>
        ) : null}

        {/* Private Comment pseudo-input */}
        {submission && (
           <TouchableOpacity style={styles.privateCommentRow} onPress={openComments}>
              <Ionicons name="person-circle-outline" size={32} color="#5F6368" />
              <View style={styles.privateCommentBox}>
                 <Text style={styles.privateCommentPlaceholder}>Add private comment</Text>
              </View>
           </TouchableOpacity>
        )}
      </View>

      {/* Plagiarism Check Modal */}
      <Modal visible={showPlagiarismModal} animationType="slide" transparent={false} onRequestClose={() => setShowPlagiarismModal(false)}>
        <View style={styles.modalHeader}>
          <TouchableOpacity onPress={() => setShowPlagiarismModal(false)} style={styles.navButton}>
            <Ionicons name="close" size={24} color={C.ink1} />
          </TouchableOpacity>
          <Text style={styles.modalTitle}>Academic Integrity Check</Text>
        </View>
        <ScrollView style={styles.plagiarismContainer}>
          {checkingPlagiarism ? (
            <View style={styles.checkingContainer}>
              <ActivityIndicator size="large" color="#0B57D0" />
              <Text style={styles.checkingText}>Analyzing document...</Text>
              <Text style={styles.checkingSubtext}>This may take a moment</Text>
            </View>
          ) : (
            <>
              {plagiarismResults.map((result, index) => (
                <View key={index} style={styles.plagiarismCard}>
                  <View style={styles.plagiarismHeaderRow}>
                    <Ionicons name="shield-checkmark" size={20} color="#0B57D0" />
                    <Text style={styles.plagiarismFileName}>{result.fileName}</Text>
                  </View>
                  {result.error ? (
                    <View style={styles.plagiarismErrorContainer}>
                      <Ionicons name="alert-circle" size={20} color="#D93025" />
                      <Text style={styles.plagiarismErrorText}>{result.error}</Text>
                    </View>
                  ) : result.originalityScore !== null ? (
                    <>
                      <View style={styles.scoreContainer}>
                        <Text style={styles.scoreLabel}>Originality Score</Text>
                        <View style={[styles.scoreCircle, result.originalityScore >= 70 ? styles.scoreCircleGood : styles.scoreCircleRisk]}>
                          <Text style={[styles.scoreValue, result.originalityScore >= 70 ? styles.scoreGood : styles.scoreRisk]}>
                            {result.originalityScore}%
                          </Text>
                        </View>
                      </View>
                      {result.isHighRisk ? (
                        <View style={styles.warningBox}>
                          <Ionicons name="warning" size={18} color="#C5221F" />
                          <Text style={styles.warningText}>
                            Low originality score detected.
                          </Text>
                        </View>
                      ) : (
                        <View style={styles.successBox}>
                          <Ionicons name="checkmark-circle" size={18} color="#137333" />
                          <Text style={styles.successText}>
                            Acceptable originality.
                          </Text>
                        </View>
                      )}
                    </>
                  ) : null}
                </View>
              ))}
              <View style={styles.plagiarismActions}>
                <TouchableOpacity style={styles.secondaryButton} onPress={() => setShowPlagiarismModal(false)}>
                  <Text style={styles.secondaryButtonText}>Cancel</Text>
                </TouchableOpacity>
                <TouchableOpacity style={styles.primaryButton} onPress={proceedWithSubmission} disabled={uploading}>
                  {uploading ? <ActivityIndicator color="#fff" /> : <Text style={styles.primaryButtonText}>Turn in anyway</Text>}
                </TouchableOpacity>
              </View>
            </>
          )}
        </ScrollView>
      </Modal>

      <FileViewerModal visible={fileViewer.visible} url={fileViewer.url} fileName={fileViewer.name} isImage={fileViewer.isImage} onClose={() => setFileViewer(v => ({ ...v, visible: false }))} />
    </View>
  );
}
